---
title: List Images
sidebarTitle: List Images
api: GET https://api.vertracloud.app/v1/images
authMethod: bearer
description: Lists the custom Docker images of the account.
---

{/* @see APICustomImage[], RESTGetAPICustomImagesResult — @vertracloud/api-types */}
Needs the `images:read` scope when called with an API key.

<ParamField query="workspace_id" type="string">
  Lists the images of that workspace instead of the account's own.
</ParamField>

### Response

<ResponseField name="response" type="array">
  An array of images.

  <Expandable title="Toggle object">

  <ResponseField name="id" type="string">
    The image ID.
  </ResponseField>

  <ResponseField name="name" type="string">
    The image name.
  </ResponseField>

  <ResponseField name="description" type="string | null">
    The image description (free text, up to 128 characters), or `null` when it has none.
  </ResponseField>

  <ResponseField name="owner" type="string">
    `account` for your own images, `system` for images provided by the platform (these cannot be changed or deleted).
  </ResponseField>

  <ResponseField name="kind" type="string">
    `own` for an image you built, or `template_copy` for the copy of a template's image that [using a template](/templates/use/using-a-template) put in your account. A copy takes one image slot and can only be used through its template.
  </ResponseField>

  <ResponseField name="origin" type="object | null">
    For a `template_copy`, the template it came from: `template_id`, `template_name` (by language) and `author_display` (`null` for a template of the platform). `null` for your own images.
  </ResponseField>

  <ResponseField name="status" type="string">
    State of the latest build: `building`, `success` or `failed`. A failed rebuild does not change the current version: apps keep running the last good one.
  </ResponseField>

  <ResponseField name="error" type="string | null">
    Readable reason of the latest failed build, or `null`.
  </ResponseField>

  <ResponseField name="digest" type="string | null">
    Digest (`sha256:...`) of the current version, or `null` while no build has succeeded.
  </ResponseField>

  <ResponseField name="uncompressed_bytes" type="integer | null">
    Size of the current version, uncompressed, in bytes.
  </ResponseField>

  <ResponseField name="layer_count" type="integer | null">
    Number of layers of the current version.
  </ResponseField>

  <ResponseField name="suggested_port" type="integer | null">
    Port suggested for a new app on this image: the first `EXPOSE` of the `Dockerfile` of the latest build context, or `80`. `null` for an image created before this field existed.
  </ResponseField>

  <ResponseField name="apps" type="object[]">
    The applications created from this image. Each has `id`, `name`, `version_id` (the image version it runs, or `null`) and `outdated` (`true` when the image has a newer version; the app moves to it on the next restart or deploy you ask for).
  </ResponseField>

  <ResponseField name="github" type="object | null">
    The linked GitHub repository and branch (`repository` as `owner/name`, `branch`), or `null` when the image is not linked.
  </ResponseField>

  <ResponseField name="current_version_id" type="string | null">
    The current version of the image, or `null` while no build has succeeded.
  </ResponseField>

  <ResponseField name="created_at" type="string" format="date-time">
    When the image was created.
  </ResponseField>

  <ResponseField name="updated_at" type="string" format="date-time">
    When the image was last updated.
  </ResponseField>

  </Expandable>
</ResponseField>

### Error Responses

<ResponseField name="code" type="string">
  The error code indicating the reason for the failure. Possible values:
  - `API_KEY_SCOPE_DENIED`: The API key does not have the required scope (`403`).
  - `INTERNAL_SERVER_ERROR`: Unexpected failure (`500`).
</ResponseField>
