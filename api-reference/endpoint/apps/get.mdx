---
title: Application Details
sidebarTitle: Get Application
api: GET https://api.vertracloud.app/v1/apps/{id}
authMethod: bearer
description: Retrieves detailed information about an application by its ID.
---

{/* @see APIApplication — @vertracloud/api-types */}
<ParamField path="id" type="string" required>
  The application ID to retrieve details. If the application belongs to a workspace, pass its ID in the `workspace_id` query parameter.
</ParamField>

<ParamField query="workspace_id" type="string">
  If the application belongs to a workspace, the workspace's ID. Required for workspace members acting on an application they don't own.
</ParamField>

### Response

<ResponseField name="response" type="object">
  The application details.

  <Expandable title="Toggle object">

  <ResponseField name="id" type="string">
    The application ID.
  </ResponseField>

  <ResponseField name="cluster" type="integer">
    The application cluster ID.
  </ResponseField>

  <ResponseField name="type" type="integer">
    The application type.
  </ResponseField>

  <ResponseField name="name" type="string">
    The application name.
  </ResponseField>

  <ResponseField name="description" type="string">
    The application description.
  </ResponseField>

  <ResponseField name="owner_id" type="string">
    The application owner ID.
  </ResponseField>

  <ResponseField name="owner_plan" type="string | null">
    The owner's plan name (for example, `pro` or `enterprise-8`), or `null` when the owner has no plan.
  </ResponseField>

  <ResponseField name="language" type="string">
    The application programming language (e.g., `javascript`).
  </ResponseField>

  <ResponseField name="ram" type="integer">
    The application RAM in MB.
  </ResponseField>

  <ResponseField name="status" type="string">
    The application status: `up`, `down` or `building` (creation in progress; most actions answer `409 APP_BUILDING`).
  </ResponseField>

  <ResponseField name="last_build_error" type="string | null">
    The code of the last build or install failure, or `null` after a successful build. One of
    `BUILD_FAILED`, `BUILD_TIMEOUT`, `BUILD_INTERRUPTED`, `INSTALL_FAILED`, `INSTALL_OUT_OF_MEMORY`,
    `INSTALL_TIMEOUT` or `INSTALL_SHIELD_BLOCKED`. The detail is in the application's logs; see
    [Errors](/api-reference/errors#build-failure-codes) for what to do about each code.
  </ResponseField>

  <ResponseField name="subdomain" type="string | null">
    The application subdomain, if applicable.
  </ResponseField>

  <ResponseField name="domains" type="APIApplicationDomain[]">
    The application's custom domains, with status, SSL state, port, path prefix and redirect. Empty when it has none.
    Each item is described in [List Application Domains](/api-reference/endpoint/apps/domains/list#response).
  </ResponseField>

  <ResponseField name="public_url" type="string | null">
    The application's canonical public URL, already computed by the server: the main custom domain
    when there is one, otherwise the `subdomain` (`null` when the app has neither, i.e. it is not web-published). Build links
    from this field, never from `subdomain` directly — a self-hosted deployment behind a
    non-standard edge may return a different host or scheme here.
  </ResponseField>

  <ResponseField name="last_snapshot" type="string | null" format="date-time">
    The timestamp of the application's last snapshot.
  </ResponseField>

  <ResponseField name="main_file" type="string">
    The main file of the application.
  </ResponseField>

  <ResponseField name="version" type="string">
    The runtime version (e.g., `recommended` or `latest`).
  </ResponseField>

  <ResponseField name="start_command" type="string | null">
    The startup command for the application, if specified.
  </ResponseField>

  <ResponseField name="build_command" type="string | null">
    The build command, if specified. Runs after dependency installation and before the
    application starts, in an isolated build environment. `null` when no build command is set.
  </ResponseField>

  <ResponseField name="offline_since" type="string | null" format="date-time">
    The timestamp since the application went offline, if applicable.
  </ResponseField>

  <ResponseField name="shield_cooldown" type="object | null">
    Present while Vertra Shield has paused the application. Includes `until` (the cooldown end time),
    `reason` (`burst` or `rate_limit`), and `direction` (`in` or `out`); `null` when no cooldown is active.
  </ResponseField>

  <ResponseField name="github" type="object | null">
    Linked GitHub repository (`repo_owner`, `repo_name`) when auto-deploy is connected; `null`
    when the app was created from a zip or snapshot. The webhook URL is not included — use
    `GET /v1/apps/{id}/deploys/webhook` for that.
  </ResponseField>

  <ResponseField name="custom_image" type="object | null">
    The custom Docker image the application runs from; `null` for a regular application. With an image, `language`, `version`, `main_file`, `start_command` and `build_command` do not apply: they come from the image. See [Custom Docker images](/guides/custom-docker-images).

    <Expandable title="custom_image properties">
      <ResponseField name="image_id" type="string">
        The image ID. `id` and `name` of this object are the application's, not the image's.
      </ResponseField>

      <ResponseField name="port" type="integer | null">
        The port the application listens on and the platform routes to; `null` for an application without web traffic.
      </ResponseField>

      <ResponseField name="version_id" type="string | null">
        The image version the application runs, or `null` when it has not started on any version yet.
      </ResponseField>

      <ResponseField name="outdated" type="boolean">
        `true` when the image has a newer version than the one the application runs. It moves to it on the next restart or deploy you ask for.
      </ResponseField>
    </Expandable>
  </ResponseField>

  <ResponseField name="template" type="object | null">
    Where the application came from when it was created from a [template](/templates/use/using-a-template), or `null` for an ordinary application.

    <Expandable title="template properties">
      <ResponseField name="template_id" type="string">The template ID.</ResponseField>
      <ResponseField name="slug" type="string">The template slug.</ResponseField>
      <ResponseField name="name" type="object">The template name by language: `pt-br`, and `en` and `es` when translated.</ResponseField>
      <ResponseField name="icon_url" type="string | null">The stable URL of the template icon, or `null` for a private template without an icon or a template that was removed.</ResponseField>
      <ResponseField name="origin" type="string">`vertra`, `community` or `private`.</ResponseField>
      <ResponseField name="version_id" type="string">The ID of the version the application was created from or last updated to.</ResponseField>
      <ResponseField name="version" type="string">That version number.</ResponseField>
      <ResponseField name="closed" type="boolean">`true` when the template is closed: the application's code is not accessible.</ResponseField>
      <ResponseField name="detached" type="boolean">`true` when the application left the template and no longer receives versions.</ResponseField>
      <ResponseField name="update_available" type="boolean">`true` when the template has a newer version the application can update to.</ResponseField>
      <ResponseField name="latest_version" type="string | null">The newer version number, or `null` when there is none.</ResponseField>
      <ResponseField name="removed" type="boolean">`true` when the author removed the template; the application keeps running but gets no more updates.</ResponseField>
      <ResponseField name="via_share_link" type="boolean">`true` when the application was created from a share link.</ResponseField>
      <ResponseField name="service_id" type="string | null">The ID of the template service this application is, or `null` when the template has a single service.</ResponseField>
      <ResponseField name="permissions" type="object">What the template lets you do with this application: `code_view`, `code_edit`, `ram_change`, `ram_below_min`, `ram_above_max`, `env_add`, `commands_edit`, `terminal`, `save_as_template` and `detach`, all booleans.</ResponseField>
      <ResponseField name="locked_deploy_settings" type="string[]">The deploy settings the template locks for this application: `zero_downtime` and/or `auto_rollback`. Changing them to another value in [`PATCH /v1/apps/:id/config`](/api-reference/endpoint/apps/updateconfig) returns `403 TEMPLATE_SETTING_LOCKED`. Empty when the application is detached.</ResponseField>
    </Expandable>
  </ResponseField>

  <ResponseField name="stack_id" type="string | null">
    The ID of the [stack](/api-reference/endpoint/stacks/get) the application belongs to, or `null` when it is not in one.
  </ResponseField>

  <ResponseField name="stack_service_key" type="string | null">
    The ID of the template service this application is inside the stack, or `null` when you added it yourself.
  </ResponseField>

  <ResponseField name="deploy" type="object">
    How a deploy replaces the running version and what happens if it fails.

    <Expandable title="deploy properties">
      <ResponseField name="zero_downtime" type="string">
        The saved choice: `off` (the default), `auto` or `on`. Change it with [Update Application Configuration](/api-reference/endpoint/apps/updateconfig).
      </ResponseField>

      <ResponseField name="zero_downtime_active" type="boolean">
        What applies to the next restart or deploy: `true` on Enterprise when `zero_downtime` is `on`, or when it is `auto` and the application receives web traffic.
      </ResponseField>

      <ResponseField name="auto_rollback" type="boolean">
        The saved choice, `false` by default. It returns the saved value on any plan, but only takes effect from Intermediary.
      </ResponseField>

      <ResponseField name="health_check_path" type="string | null">
        The HTTP path called on a new version to check that it is healthy. `null` means `/`.
      </ResponseField>
    </Expandable>
  </ResponseField>

  <ResponseField name="created_at" type="string" format="date-time">
    The timestamp when the application was created.
  </ResponseField>

  <ResponseField name="updated_at" type="string" format="date-time">
    The timestamp when the application was last updated.
  </ResponseField>

  </Expandable>
</ResponseField>

### Error Responses

<ResponseField name="code" type="string">
  The error code indicating the reason for the failure. Possible values:
  - `AUTH_TOKEN_MISSING`, `API_KEY_INVALID`, `SESSION_REVOKED`: The `Authorization` header is missing, the API key is invalid, or the dashboard session was revoked (`401`).
  - `API_KEY_IP_DENIED`, `API_KEY_SCOPE_DENIED`: The API key is used from an IP outside its allowlist, or it lacks the scope the route requires (`403`).
  - `ACCESS_DENIED`: The resource belongs to another account, or you lack the workspace permission for it (`403`).
  - `APP_NOT_FOUND`: The specified application does not exist.
</ResponseField>

<ResponseField name="message" type="string">
  A descriptive message providing additional details about the error.
</ResponseField>
