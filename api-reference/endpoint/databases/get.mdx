---
title: Get Database by ID
sidebarTitle: Get Database
api: GET https://api.vertracloud.app/v1/databases/{id}
authMethod: bearer
description: Retrieves details of a specific database by its ID.
---

{/* @see APIDatabase — @vertracloud/api-types */}
<ParamField path="id" type="string" required>
  The ID of the database to retrieve. If the database belongs to a workspace, pass its ID in the `workspace_id` query parameter.
</ParamField>

<ParamField query="workspace_id" type="string">
  If the database belongs to a workspace, the workspace's ID. Required for workspace members acting on a database they don't own.
</ParamField>

### Response

<ResponseField name="response" type="object">
  The details of the database.

  <Expandable title="Toggle object">

  <ResponseField name="id" type="string">
    The ID of the database.
  </ResponseField>

  <ResponseField name="cluster" type="number">
    The database cluster ID.
  </ResponseField>

  <ResponseField name="type" type="number">
    The type of the database.
  </ResponseField>

  <ResponseField name="name" type="string">
    The name of the database.
  </ResponseField>

  <ResponseField name="description" type="string">
    The description of the database.
  </ResponseField>

  <ResponseField name="owner_id" type="string">
    The ID of the database owner.
  </ResponseField>

  <ResponseField name="owner_plan" type="string | null">
    The owner's plan name (for example, `pro` or `enterprise-8`), or `null` when the owner has no plan.
  </ResponseField>

  <ResponseField name="status" type="string">
    The status of the database (e.g., `up`).
  </ResponseField>

  <ResponseField name="ram" type="number">
    The RAM allocated to the database in MB.
  </ResponseField>

  <ResponseField name="host" type="string">
    The host address of the database.
  </ResponseField>

  <ResponseField name="port" type="number">
    The port used by the database.
  </ResponseField>

  <ResponseField name="username" type="string">
    The user to connect with. Databases created from v26.4.8 on use `vertracloud` (default database
    `vertracloud`, not a superuser); older databases keep the engine's user (`postgres`, `default` or `root`).
  </ResponseField>

  <ResponseField name="version" type="string">
    The major version of the engine, for example `"18"` or `"8.0"`. Security updates are applied on restart without changing it.
  </ResponseField>

  <ResponseField name="created_at" type="string" format="date-time">
    The timestamp when the database was created.
  </ResponseField>

  <ResponseField name="updated_at" type="string" format="date-time">
    The timestamp when the database was last updated.
  </ResponseField>

  <ResponseField name="offline_since" type="string | null" format="date-time">
    The timestamp since the database went offline, if applicable.
  </ResponseField>

  <ResponseField name="last_snapshot" type="string | null" format="date-time">
    The timestamp of the last snapshot, if available.
  </ResponseField>

  </Expandable>
</ResponseField>

### Error Responses

<ResponseField name="code" type="string">
  The error code indicating the reason for failure. Possible values:
  - `AUTH_TOKEN_MISSING`, `API_KEY_INVALID`, `SESSION_REVOKED`: The `Authorization` header is missing, the API key is invalid, or the dashboard session was revoked (`401`).
  - `API_KEY_IP_DENIED`, `API_KEY_SCOPE_DENIED`: The API key is used from an IP outside its allowlist, or it lacks the scope the route requires (`403`).
  - `ACCESS_DENIED`: The resource belongs to another account, or you lack the workspace permission for it (`403`).
  - `DATABASE_NOT_FOUND`: The specified database does not exist or the user lacks access.
</ResponseField>

<ResponseField name="message" type="string">
  A descriptive message providing additional details about the error.
</ResponseField>
